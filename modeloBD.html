Script de la Base De Datos (SQL)

                

                 
create database db_PROYECTO_INTEGRADOR;

CREATE TABLE CUSTOMERS (
 id_customer VARCHAR(20) PRIMARY KEY,
 [name] VARCHAR(100) NOT NULL,
 email VARCHAR(100) UNIQUE NOT NULL,
 phone VARCHAR(20),
 [address] VARCHAR(150),
 registration_date DATE DEFAULT GETDATE()
);


CREATE TABLE EMPLOYEES (
    document_number INT PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    position VARCHAR(50),
    salary DECIMAL(10,2),
    phone VARCHAR(20),
    hire_date DATE DEFAULT GETDATE()
);


CREATE TABLE SUPPLIERS (
    supplier_id INT PRIMARY KEY IDENTITY(1,1),
    [name] VARCHAR(100) UNIQUE NOT NULL,
    company VARCHAR(100),
    phone VARCHAR(20),
    email VARCHAR(100),
    [address] VARCHAR(150)
    );

   
create table CATEGORIES
(
id_category INT PRIMARY KEY IDENTITY(1,1), 
[name] VARCHAR (70) NOT NULL UNIQUE,
[description] VARCHAR (100)
);

create table PRODUCTS (
id_product int PRIMARY KEY IDENTITY(1,1),
id_category int,
id_proveedor int,
nombre varchar (70) NOT NULL,
[description] varchar (100),
price decimal, 
talla varchar (20) NOT NULL,
color varchar (20),
stock int,
imagen_url varchar (200),
precio_compra decimal
CONSTRAINT FK_PRODUCTS_CATEGORIES FOREIGN KEY (id_category) REFERENCES CATEGORIES (id_category),
CONSTRAINT FK_PRODUCTS_SUPPLIERS FOREIGN KEY (id_proveedor) REFERENCES SUPPLIERS (supplier_id)
);

CREATE TABLE SALES (
    
    id_sale INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
    id_customer VARCHAR(20) NOT NULL,
    id_employee INT NOT NULL,
    sales_number NVARCHAR(50) NOT NULL,
    sale_date DATE NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    discount DECIMAL(5,2) DEFAULT 0.00,
    total DECIMAL(10,2) NOT NULL,
    payment_method NVARCHAR(50) NOT NULL,
    
    
    CONSTRAINT FK_SALES_CUSTOMERS FOREIGN KEY (id_customer) 
        REFERENCES CUSTOMERS(id_customer),
    
    CONSTRAINT FK_SALES_EMPLOYEE FOREIGN KEY (id_employee) 
        REFERENCES EMPLOYEES(document_number),
  )

CREATE TABLE SALES_DETAIL (
  
    id_detail INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
    id_sale INT NOT NULL,
    id_product INT NOT NULL,
    amount INT NOT NULL,
    unit_price DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    
    CONSTRAINT FK_SALES_DETAIL_SALES FOREIGN KEY (id_sale) REFERENCES SALES(id_sale),
    CONSTRAINT FK_SALES_DETAIL_PRODUCTS FOREIGN KEY (id_product)  REFERENCES PRODUCTS (id_product)
 );

CREATE TABLE INVENTORIES (
    id_inventory INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
    id_product INT NOT NULL,
    id_employee INT NOT NULL,
    amount INT NOT NULL,
    inventory_date DATE NOT NULL,
    
    
    CONSTRAINT FK_INVENTORIES_PRODUCTS FOREIGN KEY (id_product) REFERENCES PRODUCTS(id_product),
    
    CONSTRAINT FK_INVENTORIES_EMPLOYEES FOREIGN KEY (id_employee) REFERENCES EMPLOYEES (document_number)
        );
